&НаСервере
Процедура ОбновитьВиджетСветофорSKU()
	
	ОтчетОбъект = Отчеты.MON_СветофорSKU_ПланФакт.Создать();
	// Берем чистые настройки по умолчанию
	Настройки = ОтчетОбъект.СхемаКомпоновкиДанных.НастройкиПоУмолчанию;
	
	// --- ХАК ДЛЯ ДАШБОРДА ---
	Настройки.Структура.Очистить();
	ГруппировкаСКД = Настройки.Структура.Добавить(Тип("ГруппировкаКомпоновкиДанных"));
	ГруппировкаСКД.Использование = Истина;
	// Оставляем группировку пустой = "Детальные записи"
	
	ГруппировкаСКД.Выбор.Элементы.Очистить();
	
	// 1. Добавляем Номенклатуру
	ПолеНом = ГруппировкаСКД.Выбор.Элементы.Добавить(Тип("ВыбранноеПолеКомпоновкиДанных"));
	ПолеНом.Поле = Новый ПолеКомпоновкиДанных("Номенклатура");
	ПолеНом.Использование = Истина;
	
	// 2. Добавляем СтатусКод
	ПолеКод = ГруппировкаСКД.Выбор.Элементы.Добавить(Тип("ВыбранноеПолеКомпоновкиДанных"));
	ПолеКод.Поле = Новый ПолеКомпоновкиДанных("СтатусКод");
	ПолеКод.Использование = Истина;
	// ------------------------
	
	ДатаДляРасчета = ?(ЗначениеЗаполнено(ДатаОтчета), ДатаОтчета, ТекущаяДата());
	
	УстановитьПараметрНадежно(Настройки, "ДатаОтчета", ДатаДляРасчета);
	
	//УстановитьПараметрНадежно(Настройки, "L1_Старт",  -40);
	//УстановитьПараметрНадежно(Настройки, "L1_Финиш",  -10);
	//УстановитьПараметрНадежно(Настройки, "L2_Старт",  -90);
	//УстановитьПараметрНадежно(Настройки, "L2_Финиш",  -25);
	//УстановитьПараметрНадежно(Настройки, "U1_Старт",   100);
	//УстановитьПараметрНадежно(Настройки, "U1_Финиш",   25);
	//УстановитьПараметрНадежно(Настройки, "U2_Старт",   200);
	//УстановитьПараметрНадежно(Настройки, "U2_Финиш",   50);
	
	Попытка
		КомпоновщикМакета = Новый КомпоновщикМакетаКомпоновкиДанных;
		
		// Важно: передаем только Настройки, без пользовательских, чтобы ничего не перебилось
		МакетКомпоновки = КомпоновщикМакета.Выполнить(ОтчетОбъект.СхемаКомпоновкиДанных, Настройки, , , Тип("ГенераторМакетаКомпоновкиДанныхДляКоллекцииЗначений"));
		
		ПроцессорКомпоновки = Новый ПроцессорКомпоновкиДанных;
		ПроцессорКомпоновки.Инициализировать(МакетКомпоновки);
		
		ПроцессорВывода = Новый ПроцессорВыводаРезультатаКомпоновкиДанныхВКоллекциюЗначений;
		ТЗРезультат = Новый ТаблицаЗначений;
		ПроцессорВывода.УстановитьОбъект(ТЗРезультат);
		ПроцессорВывода.Вывести(ПроцессорКомпоновки);
		
	Исключение
		Инфо = ИнформацияОбОшибке();
		Элементы.Декорация_Заголовок.Заголовок = "Ошибка: " + Инфо.Описание;
		Возврат;
	КонецПопытки;

	Всего = ТЗРезультат.Количество();
	Красные = 0;
	Желтые = 0;
	Зеленые = 0;
	
	Для Каждого Стр Из ТЗРезультат Цикл
		Код = 1; 
		Попытка Код = Стр.СтатусКод; Исключение КонецПопытки;
		
		Если Код = 3 Тогда
			Красные = Красные + 1;
		ИначеЕсли Код = 2 Тогда
			Желтые = Желтые + 1;
		Иначе
			Зеленые = Зеленые + 1;
		КонецЕсли;
	КонецЦикла;
	
	// ВИЗУАЛИЗАЦИЯ
	Если Всего = 0 Тогда
		Элементы.Декорация_Заголовок.Заголовок = "Нет данных на эту дату";
		Элементы.Декорация_Заголовок.ЦветТекста = WebЦвета.Серый;
	Иначе
		Элементы.Декорация_Заголовок.Заголовок = "Выполнение планов SKU";
		
		Если (Красные / Всего) > 0.2 Тогда 
			Элементы.Декорация_Заголовок.ЦветТекста = WebЦвета.ТемноКрасный;
		Иначе
			Элементы.Декорация_Заголовок.ЦветТекста = WebЦвета.ТемноЗеленый;
		КонецЕсли;
	КонецЕсли;
	
	Элементы.Декорация_ИнфоВсего.Заголовок    = "Всего: " + Всего;
	Элементы.Декорация_ИнфоКрасные.Заголовок  = " " + Красные;
	Элементы.Декорация_ИнфоЖелтые.Заголовок   = " " + Желтые;
	Элементы.Декорация_ИнфоЗеленые.Заголовок  = " " + Зеленые;
	
	Элементы.Декорация_ИнфоКрасные.ЦветТекста = WebЦвета.Красный;
	Элементы.Декорация_ИнфоЖелтые.ЦветТекста  = WebЦвета.ТемноОранжевый;
	Элементы.Декорация_ИнфоЗеленые.ЦветТекста = WebЦвета.Зеленый;

КонецПроцедуры

&НаСервере
Процедура УстановитьПараметрНадежно(Настройки, ИмяПараметра, Значение)
	
	// Оставляем тот способ, который сработал в логах (через НайтиЗначениеПараметра)
	// Это самый быстрый и правильный метод.
	
	ИскомыйПараметр = Новый ПараметрКомпоновкиДанных(ИмяПараметра);
	НайденныйЭлемент = Настройки.ПараметрыДанных.НайтиЗначениеПараметра(ИскомыйПараметр);
	
	Если НайденныйЭлемент <> Неопределено Тогда
		НайденныйЭлемент.Значение = Значение;
		НайденныйЭлемент.Использование = Истина;
	КонецЕсли;

	|	Ост.ВНаличииОстаток КАК Вес,
	|	// Процент свежести
	|	ВЫБОР КОГДА Ост.Серия.ГоденДо <> ДАТАВРЕМЯ(1,1,1) И Ост.Серия.ДатаПроизводства <> ДАТАВРЕМЯ(1,1,1)
	|		И РАЗНОСТЬДАТ(Ост.Серия.ДатаПроизводства, Ост.Серия.ГоденДо, ДЕНЬ) > 0
	|	ТОГДА 
	|		(РАЗНОСТЬДАТ(&ДатаАктуальности, Ост.Серия.ГоденДо, ДЕНЬ) / РАЗНОСТЬДАТ(Ост.Серия.ДатаПроизводства, Ост.Серия.ГоденДо, ДЕНЬ)) * 100
	|	ИНАЧЕ 100 КОНЕЦ КАК ОСГ_Процент,
	|	// Срок жизни
	|	ВЫБОР КОГДА Ост.Серия.ГоденДо <> ДАТАВРЕМЯ(1, 1, 1) 
	|		ТОГДА РАЗНОСТЬДАТ(Ост.Серия.ДатаПроизводства, Ост.Серия.ГоденДо, ДЕНЬ) 
	|	ИНАЧЕ 90 КОНЕЦ КАК СрокЖизниДней

	|	ЕСТЬNULL(СУММА(ВЫБОР КОГДА ЕСТЬNULL(ВТ_Детали.ОСГ_Процент, 100) МЕЖДУ 50 И 75 ТОГДА ВТ_Детали.Вес ИНАЧЕ 0 КОНЕЦ), 0) КАК КГ_Внимание,

	|СГРУППИРОВАТЬ ПО

	|	ВТ_Планы.ПланМесяц

	|	ВТ_Детали.Склад.Наименование КАК СкладИмя,
	|	СУММА(ВЫБОР КОГДА ВТ_Детали.ОСГ_Процент < 50 ТОГДА ВТ_Детали.Вес ИНАЧЕ 0 КОНЕЦ) КАК ВесТоксичный,
	|	ВТ_Детали.Номенклатура КАК Номенклатура
	|ИЗ
	|	ВТ_Детали КАК ВТ_Детали
	|СГРУППИРОВАТЬ ПО ВТ_Детали.Склад, ВТ_Детали.Склад.Наименование, ВТ_Детали.Номенклатура";

	ТЗ_Номенклатура = РезультатПакета[3].Выгрузить(); // Итоги по товарам
	ТЗ_Склады       = РезультатПакета[4].Выгрузить(); // Итоги по складам
	
	// ---------------------------------------------------------------------
	// 2. РАСЧЕТ ИНДИКАТОРОВ (СВЕТОФОР)
	// ---------------------------------------------------------------------
	Счетчик_Дефицит = 0; Счетчик_Риск = 0; Счетчик_Норма = 0; Счетчик_Затовар = 0;
	Общий_КГ_Тухляк = 0; Общий_КГ_Распродажа = 0; Общий_КГ_Внимание = 0; Общий_КГ_Свежее = 0;

		// А. Статус КОЛИЧЕСТВА
		LifeTime = ?(Стр.СрокЖизниДней = 0, 90, Стр.СрокЖизниДней);
		ПланВМесяц = ?(Стр.ПланКГ = 0, 1, Стр.ПланКГ);
		РасходВДень = ПланВМесяц / 30;
		ОбеспеченностьДней = Стр.ОстатокКГ / РасходВДень;
		TargetDays = Мин(LifeTime * 0.4, 30);
		TargetDays = Окр(TargetDays, 1);
		
		Если ОбеспеченностьДней < 5 Тогда
			Счетчик_Дефицит = Счетчик_Дефицит + 1;
			МассивДефицитныхSKU.Добавить(Стр.Номенклатура);
		ИначеЕсли ОбеспеченностьДней < TargetDays Тогда
			Счетчик_Риск = Счетчик_Риск + 1;
		ИначеЕсли ОбеспеченностьДней > (TargetDays * 2) Тогда
			Счетчик_Затовар = Счетчик_Затовар + 1;
		Иначе
			Счетчик_Норма = Счетчик_Норма + 1;
		КонецЕсли;
		
		// Б. Суммируем КГ для плиток (Теперь тут точно Числа, а не Null)
		Общий_КГ_Тухляк     = Общий_КГ_Тухляк + Стр.КГ_Тухляк;
		Общий_КГ_Распродажа = Общий_КГ_Распродажа + Стр.КГ_Распродажа;
		Общий_КГ_Внимание   = Общий_КГ_Внимание + Стр.КГ_Внимание;
		Общий_КГ_Свежее     = Общий_КГ_Свежее + Стр.КГ_Свежее;

	// 3. ФОРМИРУЕМ ТОП-3 СПИСКИ
	// ---------------------------------------------------------------------

		Стр = ТЗ_Номенклатура[Н];
		Если Стр.КГ_Тухляк > 0 Тогда
			ТекстТопТухляк = ТекстТопТухляк + "• " + Стр.НоменклатураИмя + " (" + Формат(Стр.КГ_Тухляк/1000, "ЧДЦ=1") + "т)" + Символы.ПС;
		КонецЕсли;
	КонецЦикла;
	
	// Топ Распродажа
	ТекстТопРаспродажа = "";
	ТЗ_Номенклатура.Сортировать("КГ_Распродажа УБЫВ");
	Для Н = 0 По Мин(2, ТЗ_Номенклатура.Количество()-1) Цикл
		Стр = ТЗ_Номенклатура[Н];
		Если Стр.КГ_Распродажа > 0 Тогда
			ТекстТопРаспродажа = ТекстТопРаспродажа + "• " + Стр.НоменклатураИмя + " (" + Формат(Стр.КГ_Распродажа/1000, "ЧДЦ=1") + "т)" + Символы.ПС;
		КонецЕсли;
	КонецЦикла;
	
	// ---------------------------------------------------------------------
	// 4. НОМИНАЦИИ СКЛАДОВ
	// ---------------------------------------------------------------------
	
	// А. Склад-ТОКСИК
	ТЗ_СкладыТоксик = ТЗ_Склады.Скопировать();
	ТЗ_СкладыТоксик.Свернуть("Склад, СкладИмя", "ВесТоксичный");
	ТЗ_СкладыТоксик.Сортировать("ВесТоксичный УБЫВ");
	
	ТекстСкладТоксик = "Данных нет";
	Если ТЗ_СкладыТоксик.Количество() > 0 И ТЗ_СкладыТоксик[0].ВесТоксичный > 0 Тогда
		Лидер = ТЗ_СкладыТоксик[0];
		ТекстСкладТоксик = Лидер.СкладИмя + " (" + Формат(Лидер.ВесТоксичный/1000, "ЧДЦ=1") + "т неликвида)";
	КонецЕсли;
	
	// Б. Склад-ДЕФИЦИТ
	ТЗ_СкладыДефицит = ТЗ_Склады.Скопировать();
	ТЗ_СкладыДефицит.Колонки.Добавить("СчетчикSOS", Новый ОписаниеТипов("Число"));
	
	Для Каждого СтрСклад Из ТЗ_СкладыДефицит Цикл
		Если МассивДефицитныхSKU.Найти(СтрСклад.Номенклатура) <> Неопределено Тогда
			СтрСклад.СчетчикSOS = 1;
		КонецЕсли;
	КонецЦикла;
	
	ТЗ_СкладыДефицит.Свернуть("Склад, СкладИмя", "СчетчикSOS");
	ТЗ_СкладыДефицит.Сортировать("СчетчикSOS УБЫВ");
	
	ТекстСкладДефицит = "Данных нет";
	Если ТЗ_СкладыДефицит.Количество() > 0 И ТЗ_СкладыДефицит[0].СчетчикSOS > 0 Тогда
		Лидер = ТЗ_СкладыДефицит[0];
		ТекстСкладДефицит = Лидер.СкладИмя + " (содержит " + Лидер.СчетчикSOS + " SOS-поз.)";
	КонецЕсли;
	
	// ---------------------------------------------------------------------
	// 5. ВЫВОД НА ФОРМУ
	// ---------------------------------------------------------------------
	
	// Светофор (Штуки)
	Элементы.Дец_ЦифраДефицит.Заголовок  = Строка(Счетчик_Дефицит);
	Элементы.Дец_ЦифраРиск.Заголовок     = Строка(Счетчик_Риск);
	Элементы.Дец_ЦифраНорма.Заголовок    = Строка(Счетчик_Норма);
	Элементы.Дец_ЦифраЗатовар.Заголовок  = Строка(Счетчик_Затовар);
	
	// Качество (ТОННЫ + %)
	Попытка
		ОбщийВес = Общий_КГ_Тухляк + Общий_КГ_Распродажа + Общий_КГ_Внимание + Общий_КГ_Свежее;
		Если ОбщийВес = 0 Тогда
			ДоляТухляк=0; ДоляРаспродажа=0; ДоляВнимание=0; ДоляСвежее=0;
		Иначе
			ДоляТухляк     = (Общий_КГ_Тухляк / ОбщийВес) * 100;
			ДоляРаспродажа = (Общий_КГ_Распродажа / ОбщийВес) * 100;
			ДоляВнимание   = (Общий_КГ_Внимание / ОбщийВес) * 100;
			ДоляСвежее     = (Общий_КГ_Свежее / ОбщийВес) * 100;
		КонецЕсли;

		Элементы.Дец_КГ_Тухляк.Заголовок     = Формат(Общий_КГ_Тухляк/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляТухляк, "ЧДЦ=0") + "%)";
		Элементы.Дец_КГ_Распродажа.Заголовок = Формат(Общий_КГ_Распродажа/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляРаспродажа, "ЧДЦ=0") + "%)";
		Элементы.Дец_КГ_Внимание.Заголовок   = Формат(Общий_КГ_Внимание/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляВнимание, "ЧДЦ=0") + "%)";
		Элементы.Дец_КГ_Свежее.Заголовок     = Формат(Общий_КГ_Свежее/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляСвежее, "ЧДЦ=0") + "%)";
		
		// Новые элементы
		Элементы.Дец_ТопТухляк.Заголовок     = СокрЛП(ТекстТопТухляк);
		Элементы.Дец_ТопРаспродажа.Заголовок = СокрЛП(ТекстТопРаспродажа);
		Элементы.Дец_СкладТухляк.Заголовок   = ТекстСкладТоксик;
		Элементы.Дец_СкладДефицит.Заголовок  = ТекстСкладДефицит;

	Исключение
		Сообщить("Ошибка вывода виджета: " + ОписаниеОшибки());
	КонецПопытки;

КонецПроцедуры

&НаСервере
Функция ПодготовитьНастройкиСЛогированием()
	
	Лог = "";
	ДатаИсх = ?(ЗначениеЗаполнено(ДатаОтчета), ДатаОтчета, ТекущаяДата());
	
	ОтчетОбъект = Отчеты.MON_ДетализацияСклада.Создать();
	Компоновщик = ОтчетОбъект.КомпоновщикНастроек;
	// Загружаем дефолт
	Компоновщик.ЗагрузитьНастройки(ОтчетОбъект.СхемаКомпоновкиДанных.НастройкиПоУмолчанию);
	
	// Ищем параметр "ДатаОтчета"
	ПараметрКД = Новый ПараметрКомпоновкиДанных("ДатаОтчета");
	ЭлементОсн = Компоновщик.Настройки.ПараметрыДанных.НайтиЗначениеПараметра(ПараметрКД);
	
	Если ЭлементОсн <> Неопределено И ЭлементОсн.ИдентификаторПользовательскойНастройки <> Неопределено Тогда
		
		// Ищем в пользовательских
		ЭлементПольз = Компоновщик.ПользовательскиеНастройки.Элементы.Найти(ЭлементОсн.ИдентификаторПользовательскойНастройки);
		
		Если ЭлементПольз <> Неопределено Тогда
			ЭлементПольз.Значение = ДатаИсх;
			ЭлементПольз.Использование = Истина;
			Лог = Лог + "[ОК] Дата найдена и установлена.";
		Иначе
			// Если вдруг элемента нет - мы НЕ пытаемся его добавить (это вызовет ошибку коллекции),
			// и НЕ пытаемся использовать Фиксированные (это вызовет ошибку совпадения).
			// Просто оставляем как есть. Если галочки в конфигураторе стоят верно - мы сюда не попадем.
			Лог = Лог + "[WARN] Элемент пользовательской настройки не найден по ID.";
		КонецЕсли;
		
	КонецЕсли;
	
	Res = Новый Структура;
	Res.Вставить("Настройки", Компоновщик.ПользовательскиеНастройки);
	Res.Вставить("Лог", Лог);
	
	Возврат Res;

		TargetDays = Окр(TargetDays, 1);
		
		// Расчет Цены (Складской) и Маржи (Расчетной)
		ЦенаЗаКГ = 0; МаржаЗаКГ = 0;
		
		// Цена = Стоимость всего склада / Вес всего склада
		Если Стр.ОстатокКГ > 0 Тогда
			ЦенаЗаКГ = Стр.СтоимостьСклада / Стр.ОстатокКГ;
		КонецЕсли;
		
		// Если цены склада нет (товар весь продан), пробуем взять из продаж
		Если ЦенаЗаКГ = 0 И Стр.ПроданоКГ > 0 Тогда
			ЦенаЗаКГ = (Стр.Выручка * 0.7) / Стр.ПроданоКГ; // Эвристика: считаем, что себ-сть ~70% от выручки, если нет данных
		КонецЕсли;

		// Маржа = Средняя цена продажи - Цена склада
		Если Стр.ПроданоКГ > 0 Тогда
			СредняяЦенаПродажи = Стр.Выручка / Стр.ПроданоКГ;
			Если ЦенаЗаКГ > 0 Тогда
				МаржаЗаКГ = СредняяЦенаПродажи - ЦенаЗаКГ;
			Иначе
				МаржаЗаКГ = СредняяЦенаПродажи * 0.3; // Если себестоимости нет, берем 30% как маржу
			КонецЕсли;
		КонецЕсли;
		
		// Логика Статусов + Деньги
		Если ОбеспеченностьДней < 5 Тогда
			// SOS
			Счетчик_Дефицит = Счетчик_Дефицит + 1;
			МассивДефицитныхSKU.Добавить(Стр.Номенклатура);
			
			// Упущенная выгода
			НехваткаКГ = (TargetDays * РасходВДень) - Стр.ОстатокКГ;
			Если НехваткаКГ > 0 И МаржаЗаКГ > 0 Тогда
				Деньги_УпущенаМаржа = Деньги_УпущенаМаржа + (НехваткаКГ * МаржаЗаКГ);
			КонецЕсли;
			
		ИначеЕсли ОбеспеченностьДней < TargetDays Тогда
			Счетчик_Риск = Счетчик_Риск + 1;
		ИначеЕсли ОбеспеченностьДней > (TargetDays * 2) Тогда
			// Затовар
			Счетчик_Затовар = Счетчик_Затовар + 1;
			
			// Замороженные деньги
			ИзлишекКГ = Стр.ОстатокКГ - (TargetDays * РасходВДень);
			Если ИзлишекКГ > 0 И ЦенаЗаКГ > 0 Тогда
				Деньги_Заморожено = Деньги_Заморожено + (ИзлишекКГ * ЦенаЗаКГ);
			КонецЕсли;
		Иначе
			Счетчик_Норма = Счетчик_Норма + 1;
		КонецЕсли;
		
		// Б. Суммируем КГ по качеству
		Общий_КГ_Тухляк     = Общий_КГ_Тухляк + Стр.КГ_Тухляк;
		Общий_КГ_Распродажа = Общий_КГ_Распродажа + Стр.КГ_Распродажа;
		Общий_КГ_Внимание   = Общий_КГ_Внимание + Стр.КГ_Внимание;
		Общий_КГ_Свежее     = Общий_КГ_Свежее + Стр.КГ_Свежее;
	КонецЦикла;
	
	// 3. ТОП-3 СПИСКИ
	ТекстТопТухляк = "";
	ТЗ_Номенклатура.Сортировать("КГ_Тухляк УБЫВ");
	Для Н = 0 По Мин(2, ТЗ_Номенклатура.Количество()-1) Цикл
		Стр = ТЗ_Номенклатура[Н];
		Если Стр.КГ_Тухляк > 0 Тогда
			ТекстТопТухляк = ТекстТопТухляк + "• " + Стр.НоменклатураИмя + " (" + Формат(Стр.КГ_Тухляк/1000, "ЧДЦ=1") + "т)" + Символы.ПС;
		КонецЕсли;
	КонецЦикла;
	
	ТекстТопРаспродажа = "";
	ТЗ_Номенклатура.Сортировать("КГ_Распродажа УБЫВ");
	Для Н = 0 По Мин(2, ТЗ_Номенклатура.Количество()-1) Цикл
		Стр = ТЗ_Номенклатура[Н];
		Если Стр.КГ_Распродажа > 0 Тогда
			ТекстТопРаспродажа = ТекстТопРаспродажа + "• " + Стр.НоменклатураИмя + " (" + Формат(Стр.КГ_Распродажа/1000, "ЧДЦ=1") + "т)" + Символы.ПС;
		КонецЕсли;
	КонецЦикла;
	
	// 4. НОМИНАЦИИ СКЛАДОВ
	ТЗ_СкладыТоксик = ТЗ_Склады.Скопировать();
	ТЗ_СкладыТоксик.Свернуть("Склад, СкладИмя", "ВесТоксичный");
	ТЗ_СкладыТоксик.Сортировать("ВесТоксичный УБЫВ");
	ТекстСкладТоксик = "Данных нет";
	Если ТЗ_СкладыТоксик.Количество() > 0 И ТЗ_СкладыТоксик[0].ВесТоксичный > 0 Тогда
		Лидер = ТЗ_СкладыТоксик[0];
		ТекстСкладТоксик = Лидер.СкладИмя + " (" + Формат(Лидер.ВесТоксичный/1000, "ЧДЦ=1") + "т неликвида)";
	КонецЕсли;
	
	ТЗ_СкладыДефицит = ТЗ_Склады.Скопировать();
	ТЗ_СкладыДефицит.Колонки.Добавить("СчетчикSOS", Новый ОписаниеТипов("Число"));
	Для Каждого СтрСклад Из ТЗ_СкладыДефицит Цикл
		Если МассивДефицитныхSKU.Найти(СтрСклад.Номенклатура) <> Неопределено Тогда
			СтрСклад.СчетчикSOS = 1;
		КонецЕсли;
	КонецЦикла;
	ТЗ_СкладыДефицит.Свернуть("Склад, СкладИмя", "СчетчикSOS");
	ТЗ_СкладыДефицит.Сортировать("СчетчикSOS УБЫВ");
	ТекстСкладДефицит = "Данных нет";
	Если ТЗ_СкладыДефицит.Количество() > 0 И ТЗ_СкладыДефицит[0].СчетчикSOS > 0 Тогда
		Лидер = ТЗ_СкладыДефицит[0];
		ТекстСкладДефицит = Лидер.СкладИмя + " (содержит " + Лидер.СчетчикSOS + " SOS-поз.)";
	КонецЕсли;
	
	// 5. ВЫВОД НА ФОРМУ
	Элементы.Дец_ЦифраДефицит.Заголовок  = Строка(Счетчик_Дефицит);
	Элементы.Дец_ЦифраРиск.Заголовок     = Строка(Счетчик_Риск);
	Элементы.Дец_ЦифраНорма.Заголовок    = Строка(Счетчик_Норма);
	Элементы.Дец_ЦифраЗатовар.Заголовок  = Строка(Счетчик_Затовар);
	
	Попытка
		// ДЕНЬГИ (НОВОЕ) - Используем тыс. руб (т.р.)
		Элементы.Дец_ДеньгиЗатовар.Заголовок = "Заморожено: " + Формат(Деньги_Заморожено/1000, "ЧДЦ=0; ЧГ=3,0") + " т.р.";
		Элементы.Дец_ДеньгиДефицит.Заголовок = "Упущ. маржа: " + Формат(Деньги_УпущенаМаржа/1000, "ЧДЦ=0; ЧГ=3,0") + " т.р.";

		ОбщийВес = Общий_КГ_Тухляк + Общий_КГ_Распродажа + Общий_КГ_Внимание + Общий_КГ_Свежее;
		Если ОбщийВес = 0 Тогда
			ДоляТухляк=0; ДоляРаспродажа=0; ДоляВнимание=0; ДоляСвежее=0;
		Иначе
			ДоляТухляк     = (Общий_КГ_Тухляк / ОбщийВес) * 100;
			ДоляРаспродажа = (Общий_КГ_Распродажа / ОбщийВес) * 100;
			ДоляВнимание   = (Общий_КГ_Внимание / ОбщийВес) * 100;
			ДоляСвежее     = (Общий_КГ_Свежее / ОбщийВес) * 100;
		КонецЕсли;
		Элементы.Дец_КГ_Тухляк.Заголовок     = Формат(Общий_КГ_Тухляк/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляТухляк, "ЧДЦ=0") + "%)";
		Элементы.Дец_КГ_Распродажа.Заголовок = Формат(Общий_КГ_Распродажа/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляРаспродажа, "ЧДЦ=0") + "%)";
		Элементы.Дец_КГ_Внимание.Заголовок   = Формат(Общий_КГ_Внимание/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляВнимание, "ЧДЦ=0") + "%)";
		Элементы.Дец_КГ_Свежее.Заголовок     = Формат(Общий_КГ_Свежее/1000, "ЧДЦ=1; ЧГ=3,0") + "т (" + Формат(ДоляСвежее, "ЧДЦ=0") + "%)";
		
		Элементы.Дец_ТопТухляк.Заголовок     = СокрЛП(ТекстТопТухляк);
		Элементы.Дец_ТопРаспродажа.Заголовок = СокрЛП(ТекстТопРаспродажа);
		Элементы.Дец_СкладТухляк.Заголовок   = ТекстСкладТоксик;
		Элементы.Дец_СкладДефицит.Заголовок  = ТекстСкладДефицит;
	Исключение
		Сообщить("Ошибка вывода виджета: " + ОписаниеОшибки());
	КонецПопытки;
КонецПроцедуры

&НаСервере
Процедура ОбновитьВиджетПродажНаСервере()
	ДатаДляРасчета = ?(ЗначениеЗаполнено(ДатаОтчета), ДатаОтчета, ТекущаяДата());
	НачМес = НачалоМесяца(ДатаДляРасчета);
	КонМес = КонецМесяца(ДатаДляРасчета);

	ТабПлан = ЗапросПланПродажВиджета(НачМес, КонМес);
	ТабФакт = ЗапросФактПродажВиджета(НачМес, КонМес);
	ТабЛогистика = ЗапросЛогистикаФактВиджета(НачМес, КонМес);

	МетрикиПоОтделам = ПолучитьПустыеМетрикиПоОтделам();
	ПорогиМаржи = ПолучитьПорогиМаржиПоОтделам();

	Для Каждого Стр Из ТабПлан Цикл
		КлючОтдела = ОпределитьКлючОтдела(Стр.Подразделение);
		Если ПустаяСтрока(КлючОтдела) Тогда
			Продолжить;
		КонецЕсли;
		Метрика = МетрикиПоОтделам.Получить(КлючОтдела);
		Метрика.План = Метрика.План + ?(Стр.ПланСумма = Неопределено, 0, Стр.ПланСумма);
	КонецЦикла;

	Для Каждого Стр Из ТабФакт Цикл
		КлючОтдела = ОпределитьКлючОтдела(Стр.Подразделение);
		Если ПустаяСтрока(КлючОтдела) Тогда
			Продолжить;
		КонецЕсли;
		Метрика = МетрикиПоОтделам.Получить(КлючОтдела);
		Метрика.Факт = Метрика.Факт + ?(Стр.Выручка = Неопределено, 0, Стр.Выручка);
		Метрика.Валовая = Метрика.Валовая + ?(Стр.ВаловаяПрибыль = Неопределено, 0, Стр.ВаловаяПрибыль);
	КонецЦикла;

	Для Каждого Стр Из ТабЛогистика Цикл
		КлючОтдела = ОпределитьКлючОтдела(Стр.Подразделение);
		Если ПустаяСтрока(КлючОтдела) Тогда
			Продолжить;
		КонецЕсли;
		Метрика = МетрикиПоОтделам.Получить(КлючОтдела);
		Метрика.Логистика = Метрика.Логистика + ?(Стр.ЛогистикаФакт = Неопределено, 0, Стр.ЛогистикаФакт);
	КонецЦикла;

	ПорядокКлючей = Новый Массив;
	ПорядокКлючей.Добавить("CONS");
	ПорядокКлючей.Добавить("MEAT");
	ПорядокКлючей.Добавить("FISH");
	ПорядокКлючей.Добавить("NET");

	СуммаПлан = 0;
	СуммаФакт = 0;
	СуммаВаловаяСЛогистикой = 0;

	Для Индекс = 0 По ПорядокКлючей.Количество() - 1 Цикл
		КлючОтдела = ПорядокКлючей[Индекс];
		Метрика = МетрикиПоОтделам.Получить(КлючОтдела);
		Порог = ПорогиМаржи.Получить(КлючОтдела);
		ВаловаяСЛогистикой = Метрика.Валовая - Метрика.Логистика;

		ОборотПроц = ?(Метрика.План > 0, Метрика.Факт / Метрика.План * 100, 0);
		МаржаПроц = ?(Метрика.Факт > 0, ВаловаяСЛогистикой / Метрика.Факт * 100, 0);

		ЗаполнитьСтрокуВиджетаПродаж(Индекс + 1, Метрика.Наименование, ОборотПроц, МаржаПроц, Порог.Красный, Порог.Желтый);

		СуммаПлан = СуммаПлан + Метрика.План;
		СуммаФакт = СуммаФакт + Метрика.Факт;
		СуммаВаловаяСЛогистикой = СуммаВаловаяСЛогистикой + ВаловаяСЛогистикой;
	КонецЦикла;

&НаСервере
Функция СформироватьИтогиКонтуровЗакупокДляВиджета(ДатаДляРасчета)
	КонПериодаОтчета = КонецДня(КонецМесяца(ДатаДляРасчета));
	ДатаСрезаОстатков = НачалоДня(НачалоМесяца(ДобавитьМесяц(КонПериодаОтчета, 1)));
	СписокСкладов = ПолучитьСкладыЗакупокПоДатеОтчета(ДатаДляРасчета);
	СписокСкладовРасхода = ПолучитьСкладыРасходаЗакупокПоДатеОтчета(ДатаДляРасчета);
	ПериодСпросаНорм = ПолучитьПериодСпросаНормЗакупок(ДатаДляРасчета, СписокСкладовРасхода);
	НачПериодаСпроса = ПериодСпросаНорм.ДатаНач;
	КонПериодаСпроса = ПериодСпросаНорм.ДатаКон;
	КоличествоМесяцевСпроса = ПолучитьКоличествоМесяцевПериодаЗакупок(НачПериодаСпроса, КонПериодаСпроса);

	АгрегатыКатегорий = Новый Соответствие;
	ИтогиСпросаКонтуров = Новый Соответствие;
	ИтогиПоставокКонтуров = Новый Соответствие;

	Для Каждого Стр Из ЗапросКаноническиеОстаткиЗакупок(ДатаСрезаОстатков, СписокСкладов) Цикл
		КонтурБизнеса = ОпределитьКонтурБизнеса(Стр.Организация, Стр.Склад);
		Если КонтурБизнеса <> "Юниант" И КонтурБизнеса <> "Финист" Тогда
			Продолжить;
		КонецЕсли;

		ОрганизацияНормы = ОпределитьОрганизациюНормы(КонтурБизнеса, Стр.Склад);
		КатегорияНормы = ОпределитьКатегориюНормыДляОрганизации(ОрганизацияНормы, Стр.ВидНоменклатуры, Стр.ТоварнаяКатегория, Стр.Номенклатура);
		ТипЕдиницыКонтроля = ?(КатегорияЗакупокВСтрогихКг(КатегорияНормы), "кг", "усл.ед.");
		Итог = ПолучитьАгрегатКатегорииЗакупокВиджета(АгрегатыКатегорий, КонтурБизнеса, ОрганизацияНормы, ТипЕдиницыКонтроля, КатегорияНормы);
		Количество = ?(Стр.ВНаличииОстаток = Неопределено, 0, Стр.ВНаличииОстаток);
		Если ТипЕдиницыКонтроля = "кг" Тогда
			Итог.ОстатокКг = Итог.ОстатокКг + Количество;
		Иначе
			Итог.ОстатокУслЕд = Итог.ОстатокУслЕд + Количество;
		КонецЕсли;
		Итог.ОстатокРуб = Итог.ОстатокРуб + ?(Стр.СтоимостьОстатка = Неопределено, 0, Стр.СтоимостьОстатка);
		Итог.СтрокБезЦены = Итог.СтрокБезЦены + ?(Стр.ЦенаРубЗаКг = Неопределено Или Стр.ЦенаРубЗаКг = 0, 1, 0);
	КонецЦикла;

	Для Каждого Стр Из ЗапросПроизводственныйСпросЗакупок(НачПериодаСпроса, КонПериодаСпроса, СписокСкладовРасхода) Цикл
		КонтурБизнеса = ОпределитьКонтурБизнеса("", Стр.Склад);
		Если КонтурБизнеса <> "Юниант" И КонтурБизнеса <> "Финист" Тогда
			Продолжить;
		КонецЕсли;

		ОрганизацияНормы = ОпределитьОрганизациюНормы(КонтурБизнеса, Стр.Склад);
		КатегорияНормы = ОпределитьКатегориюНормыДляОрганизации(ОрганизацияНормы, Стр.ВидНоменклатуры, Стр.ТоварнаяКатегория, Стр.Номенклатура);
		ТипЕдиницыКонтроля = ?(КатегорияЗакупокВСтрогихКг(КатегорияНормы), "кг", "усл.ед.");
		Итог = ПолучитьАгрегатКатегорииЗакупокВиджета(АгрегатыКатегорий, КонтурБизнеса, ОрганизацияНормы, ТипЕдиницыКонтроля, КатегорияНормы);
		КоличествоСпроса = ?(Стр.РасходКоличество = Неопределено, 0, Стр.РасходКоличество);
		Итог.СпросВсего = Итог.СпросВсего + КоличествоСпроса;
		ДобавитьСуммуПоКлючуЗакупокВиджета(ИтогиСпросаКонтуров, КонтурБизнеса + "|" + ОрганизацияНормы + "|" + ТипЕдиницыКонтроля, КоличествоСпроса);
	КонецЦикла;

	Для Каждого Стр Из ЗапросПроизводственныйСпросЗакупокПоМесяцам(НачПериодаСпроса, КонПериодаСпроса, СписокСкладовРасхода) Цикл
		КонтурБизнеса = ОпределитьКонтурБизнеса("", Стр.Склад);
		Если КонтурБизнеса <> "Юниант" И КонтурБизнеса <> "Финист" Тогда
			Продолжить;
		КонецЕсли;

		ОрганизацияНормы = ОпределитьОрганизациюНормы(КонтурБизнеса, Стр.Склад);
		КатегорияНормы = ОпределитьКатегориюНормыДляОрганизации(ОрганизацияНормы, Стр.ВидНоменклатуры, Стр.ТоварнаяКатегория, Стр.Номенклатура);
		ТипЕдиницыКонтроля = ?(КатегорияЗакупокВСтрогихКг(КатегорияНормы), "кг", "усл.ед.");
		Итог = ПолучитьАгрегатКатегорииЗакупокВиджета(АгрегатыКатегорий, КонтурБизнеса, ОрганизацияНормы, ТипЕдиницыКонтроля, КатегорияНормы);
		КлючМесяца = Год(НачалоМесяца(Стр.МесяцСпроса)) * 100 + Месяц(НачалоМесяца(Стр.МесяцСпроса));
		ДобавитьСуммуПоКлючуЗакупокВиджета(Итог.МесяцыСпроса, КлючМесяца, ?(Стр.РасходКоличество = Неопределено, 0, Стр.РасходКоличество));
	КонецЦикла;

	Для Каждого Стр Из ЗапросПоставкиЗакупокПоМесяцам(НачПериодаСпроса, КонПериодаСпроса) Цикл
		КонтурБизнеса = ОпределитьКонтурБизнеса(Стр.Организация, "");
		Если КонтурБизнеса <> "Юниант" И КонтурБизнеса <> "Финист" Тогда
			Продолжить;
		КонецЕсли;

		ОрганизацияНормы = ОпределитьОрганизациюНормы(КонтурБизнеса, "");
		КатегорияНормы = ОпределитьКатегориюНормыДляОрганизации(ОрганизацияНормы, Стр.ВидНоменклатуры, Стр.ТоварнаяКатегория, Стр.Номенклатура);
		ТипЕдиницыКонтроля = ?(КатегорияЗакупокВСтрогихКг(КатегорияНормы), "кг", "усл.ед.");
		Итог = ПолучитьАгрегатКатегорииЗакупокВиджета(АгрегатыКатегорий, КонтурБизнеса, ОрганизацияНормы, ТипЕдиницыКонтроля, КатегорияНормы);
		КоличествоПоставки = ?(Стр.КоличествоПоставки = Неопределено, 0, Стр.КоличествоПоставки);
		СтоимостьПоставки = ?(Стр.СтоимостьПоставки = Неопределено, 0, Стр.СтоимостьПоставки);
		ДокументовПоставки = ?(Стр.ДокументовПоставки = Неопределено, 0, Стр.ДокументовПоставки);
		КлючМесяца = Год(НачалоМесяца(Стр.МесяцПоставки)) * 100 + Месяц(НачалоМесяца(Стр.МесяцПоставки));
		ДобавитьСуммуПоКлючуЗакупокВиджета(Итог.МесяцыПоставок, КлючМесяца, КоличествоПоставки);
		Итог.ПоставкаВсего = Итог.ПоставкаВсего + КоличествоПоставки;
		Итог.СтоимостьПоставки = Итог.СтоимостьПоставки + СтоимостьПоставки;
		Итог.ПоставокДокументов = Итог.ПоставокДокументов + ДокументовПоставки;
		ДобавитьСуммуПоКлючуЗакупокВиджета(ИтогиПоставокКонтуров, КонтурБизнеса + "|" + ОрганизацияНормы + "|" + ТипЕдиницыКонтроля, КоличествоПоставки);
	КонецЦикла;

	ИтогиКонтуров = Новый Соответствие;
	ИтогиКонтуров.Вставить("Юниант", СоздатьИтогКонтураЗакупокВиджета("Юниант"));
	ИтогиКонтуров.Вставить("Финист", СоздатьИтогКонтураЗакупокВиджета("Финист"));

	Для Каждого ЭлементКатегории Из АгрегатыКатегорий Цикл
		Категория = ЭлементКатегории.Значение;
		Если Категория.СпросВсего = 0 Тогда
			Продолжить;
		КонецЕсли;

		СпросВМесяц = Категория.СпросВсего / КоличествоМесяцевСпроса;
		СпросВДень = Окр(СпросВМесяц / 30, 2);
		АктивныхМесяцев = 0;
		МаксимальныйИндекс = 0;
		Для Каждого ЭлементМесяца Из Категория.МесяцыСпроса Цикл
			СпросМесяца = ЭлементМесяца.Значение;
			Если СпросМесяца > 0 Тогда
				АктивныхМесяцев = АктивныхМесяцев + 1;
			КонецЕсли;
			ИндексСезонности = ?(СпросВМесяц > 0, СпросМесяца / СпросВМесяц, 0);
			Если ИндексСезонности > МаксимальныйИндекс Тогда
				МаксимальныйИндекс = ИндексСезонности;
			КонецЕсли;
		КонецЦикла;

		МесяцевСПоставками = 0;
		МаксимальнаяПартия = 0;
		Для Каждого ЭлементМесяца Из Категория.МесяцыПоставок Цикл
			ПоставкаМесяца = ЭлементМесяца.Значение;
			Если ПоставкаМесяца > 0 Тогда
				МесяцевСПоставками = МесяцевСПоставками + 1;
			КонецЕсли;
			Если ПоставкаМесяца > МаксимальнаяПартия Тогда
				МаксимальнаяПартия = ПоставкаМесяца;
			КонецЕсли;
		КонецЦикла;

		КлючКонтурТип = Категория.КонтурБизнеса + "|" + Категория.ОрганизацияНормы + "|" + Категория.ТипЕдиницыКонтроля;
		СпросКонтура = ?(ИтогиСпросаКонтуров.Получить(КлючКонтурТип) = Неопределено, 0, ИтогиСпросаКонтуров.Получить(КлючКонтурТип));
		ПоставкаКонтура = ?(ИтогиПоставокКонтуров.Получить(КлючКонтурТип) = Неопределено, 0, ИтогиПоставокКонтуров.Получить(КлючКонтурТип));
		ДоляСпроса = ?(СпросКонтура > 0, Категория.СпросВсего / СпросКонтура, 0);
		ДоляПоставки = ?(ПоставкаКонтура > 0, Категория.ПоставкаВсего / ПоставкаКонтура, 0);
		КлассЗначимостиСпроса = ПолучитьКлассЗначимостиСпросаЗакупок(ДоляСпроса);
		СтатусСезонности = ПолучитьСтатусСезонностиЗакупок(МаксимальныйИндекс, АктивныхМесяцев, КоличествоМесяцевСпроса, КлассЗначимостиСпроса);
		ТипПолитики = ПолучитьТипПолитикиПоСезонностиЗакупок(СтатусСезонности, МаксимальныйИндекс, АктивныхМесяцев, КоличествоМесяцевСпроса);
		КлассПоставки = ПолучитьКлассПоставкиЗакупок(МесяцевСПоставками, КоличествоМесяцевСпроса, ДоляПоставки);
		ПериодПополненияДней = ?(МесяцевСПоставками > 0, Окр(КоличествоМесяцевСпроса * 30 / МесяцевСПоставками, 0), 0);
		СредняяПартия = ?(Категория.ПоставокДокументов > 0, Категория.ПоставкаВсего / Категория.ПоставокДокументов, Категория.ПоставкаВсего);
		МинимальнаяПартия = ?(СредняяПартия > 0, СредняяПартия, МаксимальнаяПартия);
		БуферПоставки = ?(МаксимальнаяПартия > МинимальнаяПартия, Окр(МаксимальнаяПартия - МинимальнаяПартия, 2), 0);
		СтраховойЗапасДней = ПолучитьРасчетныйСтраховойЗапасДнейЗакупок(ТипПолитики, КлассЗначимостиСпроса);
		ПиковыйБуфер = Окр(ПолучитьПиковыйБуферНормыЗакупок(СпросВМесяц, МаксимальныйИндекс, ТипПолитики, КлассЗначимостиСпроса), 2);
		ПериодПополненияДней = ОграничитьПериодПополненияПоЗрелостиКонтураЗакупок(Категория.КонтурБизнеса, ТипПолитики, ПериодПополненияДней);
		ПиковыйБуфер = ОграничитьПиковыйБуферПоЗрелостиКонтураЗакупок(Категория.КонтурБизнеса, ПиковыйБуфер, СпросВМесяц);

		Если ЭтоХвостоваяПолитикаНормыЗакупок(ТипПолитики, КлассЗначимостиСпроса) Тогда
			НормаМин = 0;
			НормаЦель = 0;
			НормаМакс = 0;
			СтатусРасчетаНормы = "Не нормируется: хвост";
		Иначе
			НормаМин = Окр(СпросВДень * СтраховойЗапасДней, 2);
			НормаЦель = Окр(СпросВДень * (ПериодПополненияДней + СтраховойЗапасДней) + ПиковыйБуфер, 2);
			НормаЦель = ОграничитьНормуЦельПоЗрелостиКонтураЗакупок(
				Категория.КонтурБизнеса,
				ТипПолитики,
				КлассЗначимостиСпроса,
				НормаЦель,
				СпросВМесяц);
			НормаМакс = Окр(НормаЦель + МинимальнаяПартия + БуферПоставки, 2);
			СтатусРасчетаНормы = ПолучитьСтатусРасчетаБазыНормыЗакупок(ТипПолитики, КлассЗначимостиСпроса, КлассПоставки, 0, ПериодПополненияДней);
		КонецЕсли;

		ОстатокКонтроль = ?(Категория.ТипЕдиницыКонтроля = "кг", Категория.ОстатокКг, Категория.ОстатокУслЕд);
		ЦенаОстатка = ?(ОстатокКонтроль > 0, Окр(Категория.ОстатокРуб / ОстатокКонтроль, 2), 0);
		ЦенаПоставки = ?(Категория.ПоставкаВсего > 0 И Категория.СтоимостьПоставки > 0, Окр(Категория.СтоимостьПоставки / Категория.ПоставкаВсего, 2), 0);
		ЦенаКонтроль = ?(ЦенаПоставки > 0, ЦенаПоставки, ЦенаОстатка);
		НормаЦельРуб = Окр(НормаЦель * ЦенаКонтроль, 2);
		СтатусКатегории = ПолучитьСтатусКоридораНормыЗакупок(ОстатокКонтроль, НормаМин, НормаЦель, НормаМакс, СтатусРасчетаНормы);

		ИтогКонтура = ИтогиКонтуров.Получить(Категория.КонтурБизнеса);
		ИтогКонтура.ОстатокРуб = ИтогКонтура.ОстатокРуб + Категория.ОстатокРуб;
		НормаУчаствует = Не (СтатусКатегории = "Не нормируется" Или СтатусКатегории = "Нет нормы");
		ИтогКонтура.НормаРуб = ИтогКонтура.НормаРуб + ?(НормаУчаствует, НормаЦельРуб, 0);
		Если Категория.ТипЕдиницыКонтроля = "кг" Тогда
			ИтогКонтура.ОстатокКг = ИтогКонтура.ОстатокКг + ОстатокКонтроль;
			ИтогКонтура.НормаКг = ИтогКонтура.НормаКг + ?(НормаУчаствует, НормаЦель, 0);
		Иначе
			ИтогКонтура.ОстатокУЕ = ИтогКонтура.ОстатокУЕ + ОстатокКонтроль;
			ИтогКонтура.НормаУЕ = ИтогКонтура.НормаУЕ + ?(НормаУчаствует, НормаЦель, 0);
		КонецЕсли;
		УвеличитьСчетчикСтатусаКатегорииЗакупокВиджета(ИтогКонтура, СтатусКатегории);
	КонецЦикла;

	Для Каждого ЭлементКонтура Из ИтогиКонтуров Цикл
		ЗаполнитьСтатусыИтогаКонтураЗакупокВиджета(ЭлементКонтура.Значение);
	КонецЦикла;

	Возврат ИтогиКонтуров;

&НаСервере
Процедура УвеличитьСчетчикСтатусаКатегорииЗакупокВиджета(ИтогКонтура, СтатусКатегории)
	Если СтатусКатегории = "Дефицит" Тогда
		ИтогКонтура.КатегорийДефицит = ИтогКонтура.КатегорийДефицит + 1;
	ИначеЕсли СтатусКатегории = "Ниже цели" Тогда
		ИтогКонтура.КатегорийНижеЦели = ИтогКонтура.КатегорийНижеЦели + 1;
	ИначеЕсли СтатусКатегории = "Избыток" Тогда
		ИтогКонтура.КатегорийИзбыток = ИтогКонтура.КатегорийИзбыток + 1;
	ИначеЕсли СтатусКатегории = "Нет нормы" Тогда
		ИтогКонтура.КатегорийНетНормы = ИтогКонтура.КатегорийНетНормы + 1;
	ИначеЕсли СтатусКатегории = "Не нормируется" Тогда
		ИтогКонтура.КатегорийНеНормируется = ИтогКонтура.КатегорийНеНормируется + 1;

&НаСервере
Процедура ЗаполнитьСтатусыИтогаКонтураЗакупокВиджета(ИтогКонтура)
	ИтогКонтура.ОтклонениеРуб = Окр(ИтогКонтура.ОстатокРуб - ИтогКонтура.НормаРуб, 2);
	ИтогКонтура.ПокрытиеРуб = ?(ИтогКонтура.НормаРуб > 0, Окр(ИтогКонтура.ОстатокРуб / ИтогКонтура.НормаРуб, 2), 0);
	ИтогКонтура.СтатусРуб = ПолучитьСтатусПокрытияКонтураЗакупок(ИтогКонтура.ПокрытиеРуб, ИтогКонтура.НормаРуб);
	ИтогКонтура.РекомендацияРуб = ПолучитьРекомендациюМетрикиЗакупок(ИтогКонтура.СтатусРуб, "руб");

	ИтогКонтура.ОтклонениеКг = Окр(ИтогКонтура.ОстатокКг - ИтогКонтура.НормаКг, 2);
	ИтогКонтура.ПокрытиеКг = ?(ИтогКонтура.НормаКг > 0, Окр(ИтогКонтура.ОстатокКг / ИтогКонтура.НормаКг, 2), 0);
	ИтогКонтура.СтатусКг = ПолучитьСтатусПокрытияКонтураЗакупок(ИтогКонтура.ПокрытиеКг, ИтогКонтура.НормаКг);

Функция ПолучитьПериодСпросаНормЗакупок(ДатаОтчета, СписокСкладовРасхода)
	НачалоНормативногоОкна = Дата(2025, 1, 1);
	ПоследняяДатаДанных = ПолучитьПоследнююДатуДанныхСпросаЗакупок(НачалоНормативногоОкна, КонецДня(ТекущаяДата()), СписокСкладовРасхода);
	Если ПоследняяДатаДанных = Неопределено Тогда
		КонецНормативногоОкна = КонецДня(Дата(2025, 12, 31));
		Комментарий = "Референс нормы: 2025 год; последняя дата данных спроса не определена";
	Иначе
		Если День(ПоследняяДатаДанных) < День(КонецМесяца(ПоследняяДатаДанных)) Тогда
			КонецНормативногоОкна = КонецДня(КонецМесяца(ДобавитьМесяц(ПоследняяДатаДанных, -1)));
		Иначе
			КонецНормативногоОкна = КонецДня(КонецМесяца(ПоследняяДатаДанных));
		КонецЕсли;

		Если КонецНормативногоОкна < НачалоНормативногоОкна Тогда
			КонецНормативногоОкна = КонецДня(КонецМесяца(НачалоНормативногоОкна));
		КонецЕсли;

		Комментарий = "Референс нормы: с 01.01.2025 по последний закрытый месяц данных спроса; последняя дата данных="
			+ Формат(ПоследняяДатаДанных, "ДФ=dd.MM.yyyy");
	КонецЕсли;

	Возврат Новый Структура("ДатаНач,ДатаКон,Комментарий",
		НачалоНормативногоОкна,
		КонецНормативногоОкна,
		Комментарий);

Функция ПолучитьСтатусСезонностиЗакупок(МаксимальныйИндекс, АктивныхМесяцев, МесяцевВПериоде, КлассЗначимостиСпроса)
	Если МесяцевВПериоде < 6 Тогда
		Возврат "Недостаточно истории";
	КонецЕсли;

	Если КлассЗначимостиСпроса = "Хвост" Тогда
		Возврат "Хвостовой спрос";
	ИначеЕсли КлассЗначимостиСпроса = "Малая" И АктивныхМесяцев <= Окр(МесяцевВПериоде * 0.55, 0) Тогда
		Возврат "Малый нерегулярный спрос";
	КонецЕсли;

	ПорогСезонногоОкна = МаксимумЗакупок(2, Окр(МесяцевВПериоде * 0.35, 0));
	ПорогСжатогоСезона = МаксимумЗакупок(3, Окр(МесяцевВПериоде * 0.55, 0));
	ПорогРегулярногоСпроса = Окр(МесяцевВПериоде * 0.8, 0);

	Если АктивныхМесяцев <= ПорогСезонногоОкна И МаксимальныйИндекс >= 2 Тогда
		Возврат "Выраженное сезонное окно";
	ИначеЕсли АктивныхМесяцев <= ПорогСжатогоСезона Тогда
		Возврат "Сжатый сезон";
	ИначеЕсли АктивныхМесяцев >= ПорогРегулярногоСпроса И МаксимальныйИндекс >= 1.75 Тогда
		Возврат "Регулярный спрос с пиками";
	ИначеЕсли МаксимальныйИндекс >= 1.75 Тогда
		Возврат "Есть выраженные пики";
	ИначеЕсли АктивныхМесяцев >= ПорогРегулярногоСпроса И МаксимальныйИндекс >= 1.35 Тогда
		Возврат "Регулярный спрос с умеренными пиками";
	ИначеЕсли МаксимальныйИндекс >= 1.35 Тогда
		Возврат "Умеренная сезонность";
	КонецЕсли;

	Возврат "Ровный спрос";
