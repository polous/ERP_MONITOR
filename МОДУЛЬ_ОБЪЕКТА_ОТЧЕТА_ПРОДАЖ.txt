Процедура ПриКомпоновкеРезультата(ДокументРезультата, ДанныеРасшифровки, СтандартнаяОбработка)
	
	СтандартнаяОбработка = Ложь;
	
	ДатаОтчета = ПолучитьПараметрСКД("ДатаОтчета", ТекущаяДата());
	ТабРезультат = СформироватьДанные(ДатаОтчета);
	
	ВнешниеНаборы = Новый Структура;
	ВнешниеНаборы.Вставить("МониторПродаж", ТабРезультат);
	
	КомпоновщикМакета = Новый КомпоновщикМакетаКомпоновкиДанных;
	МакетКомпоновки = КомпоновщикМакета.Выполнить(
		СхемаКомпоновкиДанных,
		КомпоновщикНастроек.Настройки,
		ДанныеРасшифровки,
		,
		Тип("ГенераторМакетаКомпоновкиДанных")
	);
	
	ПроцессорКомпоновки = Новый ПроцессорКомпоновкиДанных;
	ПроцессорКомпоновки.Инициализировать(МакетКомпоновки, ВнешниеНаборы, ДанныеРасшифровки);
	
	ПроцессорВывода = Новый ПроцессорВыводаРезультатаКомпоновкиДанныхВТабличныйДокумент;
	ПроцессорВывода.УстановитьДокумент(ДокументРезультата);
	ПроцессорВывода.Вывести(ПроцессорКомпоновки, Истина);
	
КонецПроцедуры

Функция ПолучитьПараметрСКД(ИмяПараметра, ЗначениеПоУмолчанию = Неопределено)
	Попытка
		ПараметрКД = КомпоновкаДанныхКлиентСервер.ПолучитьПараметр(КомпоновщикНастроек, ИмяПараметра);
		Если ПараметрКД <> Неопределено И ПараметрКД.Значение <> Неопределено Тогда
			Возврат ПараметрКД.Значение;
		КонецЕсли;
	Исключение
	КонецПопытки;
	
	Попытка
		Для Каждого ЭлементПольз Из КомпоновщикНастроек.ПользовательскиеНастройки.Элементы Цикл
			Попытка
				Если ЭлементПольз.Параметр <> Неопределено
					И ЭлементПольз.Параметр.Имя = ИмяПараметра
					И ЭлементПольз.Значение <> Неопределено Тогда
					Возврат ЭлементПольз.Значение;
				КонецЕсли;
			Исключение
			КонецПопытки;
		КонецЦикла;
	Исключение
	КонецПопытки;
	
	Попытка
		Для Каждого ЭлементНастроек Из КомпоновщикНастроек.Настройки.ПараметрыДанных.Элементы Цикл
			Попытка
				Если ЭлементНастроек.Параметр <> Неопределено
					И ЭлементНастроек.Параметр.Имя = ИмяПараметра
					И ЭлементНастроек.Значение <> Неопределено Тогда
					Возврат ЭлементНастроек.Значение;
				КонецЕсли;
			Исключение
			КонецПопытки;
		КонецЦикла;
	Исключение
	КонецПопытки;
	
	ПараметрКД = Новый ПараметрКомпоновкиДанных(ИмяПараметра);
	ЭлементОсн = КомпоновщикНастроек.Настройки.ПараметрыДанных.НайтиЗначениеПараметра(ПараметрКД);
	
	Если ЭлементОсн = Неопределено Тогда
		Возврат ЗначениеПоУмолчанию;
	КонецЕсли;
	
	Если ЭлементОсн.ИдентификаторПользовательскойНастройки <> Неопределено Тогда
		ЭлементПольз = КомпоновщикНастроек.ПользовательскиеНастройки.Элементы.Найти(ЭлементОсн.ИдентификаторПользовательскойНастройки);
		Если ЭлементПольз <> Неопределено Тогда
			Если ЭлементПольз.Значение <> Неопределено Тогда
				Возврат ЭлементПольз.Значение;
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;
	
	Если ЭлементОсн.Значение <> Неопределено Тогда
		Возврат ЭлементОсн.Значение;
	КонецЕсли;
	
	Возврат ЗначениеПоУмолчанию;
КонецФункции

Функция СформироватьДанные(ДатаРасчета) Экспорт
	
	ДатаНач = НачалоДня(НачалоМесяца(ДатаРасчета));
	ДатаКон = КонецДня(КонецМесяца(ДатаРасчета));
	
	Дерево = Новый Соответствие; // Хранилище: Подразделение -> Менеджеры
	
	// 1. Инициализация всех целевых менеджеров и подразделений
	ТабМенеджеры = ЗапросМенеджеры();
	Для Каждого Стр Из ТабМенеджеры Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, Стр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			ПолучитьУзелМенеджера(УзелПодр, Стр.Менеджер);
		КонецЕсли;
	КонецЦикла;
	
	// 2. Сбор суммируемых показателей (План, Факт)
	ТабПлан = ЗапросПлан(ДатаНач, ДатаКон);
	Для Каждого Стр Из ТабПлан Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, Стр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			УзелПодр.ПродажиРубПлан = УзелПодр.ПродажиРубПлан + Стр.ПланСумма;
			УзелМен = ПолучитьУзелМенеджера(УзелПодр, Стр.Менеджер);
			УзелМен.ПродажиРубПлан = УзелМен.ПродажиРубПлан + Стр.ПланСумма;
		КонецЕсли;
	КонецЦикла;
	
	ТабПланКг = ЗапросПланКг(ДатаНач, ДатаКон);
	Для Каждого Стр Из ТабПланКг Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, Стр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			УзелПодр.ПродажиКгПлан = УзелПодр.ПродажиКгПлан + Стр.КгПлан;
			УзелМен = ПолучитьУзелМенеджера(УзелПодр, Стр.Менеджер);
			УзелМен.ПродажиКгПлан = УзелМен.ПродажиКгПлан + Стр.КгПлан;
		КонецЕсли;
	КонецЦикла;
	
	ТабФакт = ЗапросФакт(ДатаНач, ДатаКон);
	Для Каждого Стр Из ТабФакт Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, Стр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			УзелПодр.ПродажиРубФакт = УзелПодр.ПродажиРубФакт + Стр.Выручка;
			УзелПодр.ПродажиКгФакт = УзелПодр.ПродажиКгФакт + Стр.КгФакт;
			УзелПодр.Себестоимость = УзелПодр.Себестоимость + Стр.Себестоимость;
			УзелПодр.ВаловаяПрибыль = УзелПодр.ВаловаяПрибыль + Стр.ВаловаяПрибыль;
			
			УзелМен = ПолучитьУзелМенеджера(УзелПодр, Стр.Менеджер);
			УзелМен.ПродажиРубФакт = УзелМен.ПродажиРубФакт + Стр.Выручка;
			УзелМен.ПродажиКгФакт = УзелМен.ПродажиКгФакт + Стр.КгФакт;
			УзелМен.Себестоимость = УзелМен.Себестоимость + Стр.Себестоимость;
			УзелМен.ВаловаяПрибыль = УзелМен.ВаловаяПрибыль + Стр.ВаловаяПрибыль;
		КонецЕсли;
	КонецЦикла;
	
	ТабЛогистика = ЗапросЛогистикаФакт(ДатаНач, ДатаКон);
	Для Каждого Стр Из ТабЛогистика Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, Стр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			УзелПодр.ЛогистикаФакт = УзелПодр.ЛогистикаФакт + Стр.ЛогистикаФакт;
			УзелМен = ПолучитьУзелМенеджера(УзелПодр, Стр.Менеджер);
			УзелМен.ЛогистикаФакт = УзелМен.ЛогистикаФакт + Стр.ЛогистикаФакт;
		КонецЕсли;
	КонецЦикла;
	
	КэшПДЗ = ПолучитьКэшДанныхПДЗ(НачалоДня(ДобавитьМесяц(ДатаКон, -6)), ДатаКон);
	ТабПДЗ = РассчитатьПДЗПоМенеджерам(ДатаКон, КэшПДЗ);
	Для Каждого Стр Из ТабПДЗ Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, Стр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			УзелПодр.ПДЗ = УзелПодр.ПДЗ + Стр.ПДЗ;
			УзелМен = ПолучитьУзелМенеджера(УзелПодр, Стр.Менеджер);
			УзелМен.ПДЗ = УзелМен.ПДЗ + Стр.ПДЗ;
		КонецЕсли;
	КонецЦикла;
	
	// 3. Сбор несуммируемых показателей (Уникальные клиенты - АКБ)
	РезАКБ = ЗапросАКБ(ДатаНач, ДатаКон);
	ВыбАКБПодр = РезАКБ.Выбрать(ОбходРезультатаЗапроса.ПоГруппировкам, "Подразделение");
	Пока ВыбАКБПодр.Следующий() Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, ВыбАКБПодр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			УзелПодр.АКБМесяц = ВыбАКБПодр.АКБМесяц;
			ВыбАКБМен = ВыбАКБПодр.Выбрать(ОбходРезультатаЗапроса.ПоГруппировкам, "Менеджер");
			Пока ВыбАКБМен.Следующий() Цикл
				УзелМен = ПолучитьУзелМенеджера(УзелПодр, ВыбАКБМен.Менеджер);
				УзелМен.АКБМесяц = ВыбАКБМен.АКБМесяц;
			КонецЦикла;
		КонецЕсли;
	КонецЦикла;
	
	// 4. Новые клиенты (Тоже уникальные)
	РезНовКл = ЗапросНовыеКлиенты(ДатаНач, ДатаКон);
	ВыбНовКлПодр = РезНовКл.Выбрать(ОбходРезультатаЗапроса.ПоГруппировкам, "Подразделение");
	Пока ВыбНовКлПодр.Следующий() Цикл
		УзелПодр = ПолучитьУзелПодразделения(Дерево, ВыбНовКлПодр.Подразделение);
		Если УзелПодр <> Неопределено Тогда
			УзелПодр.НовыеКлиенты = ВыбНовКлПодр.НовыеКлиенты;
			ВыбНовКлМен = ВыбНовКлПодр.Выбрать(ОбходРезультатаЗапроса.ПоГруппировкам, "Менеджер");
			Пока ВыбНовКлМен.Следующий() Цикл
				УзелМен = ПолучитьУзелМенеджера(УзелПодр, ВыбНовКлМен.Менеджер);
				УзелМен.НовыеКлиенты = ВыбНовКлМен.НовыеКлиенты;
			КонецЦикла;
		КонецЕсли;
	КонецЦикла;
	
	// 5. Новые сетевые позиции
	ЗаполнитьНовыеСетевыеПозиции(ДатаНач, ДатаКон, Дерево);
	
	// 6. Формирование таблицы данных (только детальные записи менеджеров)
	ТабРезультат = Новый ТаблицаЗначений;
	ТабРезультат.Колонки.Добавить("Подразделение");
	ТабРезультат.Колонки.Добавить("ПодразделениеНаименование", Новый ОписаниеТипов("Строка"));
	ТабРезультат.Колонки.Добавить("Менеджер");
	ТабРезультат.Колонки.Добавить("МенеджерНаименование", Новый ОписаниеТипов("Строка"));
	
	ТабРезультат.Колонки.Добавить("ПродажиРубПлан", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("ПродажиРубФакт", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("ПродажиКгПлан", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("ПродажиКгФакт", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("Себестоимость", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("ВаловаяПрибыль", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("ЛогистикаФакт", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("ВаловаяПрибыльСЛогистикой", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("ПДЗ", Новый ОписаниеТипов("Число"));
	
	ТабРезультат.Колонки.Добавить("АКБМесяц", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("НовыеКлиенты", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("НовыеСетевыеПозиции", Новый ОписаниеТипов("Число"));
	
	// Скрытые колонки для итогов группировки СКД
	ТабРезультат.Колонки.Добавить("АКБМесяц_Подр", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("НовыеКлиенты_Подр", Новый ОписаниеТипов("Число"));
	ТабРезультат.Колонки.Добавить("НовыеСетевыеПозиции_Подр", Новый ОписаниеТипов("Число"));
	
	Для Каждого КлючЗначение Из Дерево Цикл
		УзелПодр = КлючЗначение.Значение;
		
		Для Каждого КлючЗначениеМен Из УзелПодр.Менеджеры Цикл
			УзелМен = КлючЗначениеМен.Значение;

			Если УзелМен.ПродажиРубПлан = 0
				И УзелМен.ПродажиРубФакт = 0
				И УзелМен.ПродажиКгПлан = 0
				И УзелМен.ПродажиКгФакт = 0
				И УзелМен.Себестоимость = 0
				И УзелМен.ВаловаяПрибыль = 0
				И УзелМен.ЛогистикаФакт = 0
				И УзелМен.ПДЗ = 0
				И УзелМен.АКБМесяц = 0
				И УзелМен.НовыеКлиенты = 0
				И УзелМен.НовыеСетевыеПозиции = 0 Тогда
				Продолжить;
			КонецЕсли;

			СтрМен = ТабРезультат.Добавить();
			
			СтрМен.Подразделение = УзелПодр.Подразделение;
			СтрМен.ПодразделениеНаименование = Строка(УзелПодр.Подразделение);
			СтрМен.Менеджер = УзелМен.Менеджер;
			СтрМен.МенеджерНаименование = Строка(УзелМен.Менеджер);
			
			СтрМен.ПродажиРубПлан = УзелМен.ПродажиРубПлан;
			СтрМен.ПродажиРубФакт = УзелМен.ПродажиРубФакт;
			СтрМен.ПродажиКгПлан = УзелМен.ПродажиКгПлан;
			СтрМен.ПродажиКгФакт = УзелМен.ПродажиКгФакт;
			СтрМен.Себестоимость = УзелМен.Себестоимость;
			СтрМен.ВаловаяПрибыль = УзелМен.ВаловаяПрибыль;
			СтрМен.ЛогистикаФакт = УзелМен.ЛогистикаФакт;
			СтрМен.ВаловаяПрибыльСЛогистикой = УзелМен.ВаловаяПрибыль - УзелМен.ЛогистикаФакт;
			СтрМен.ПДЗ = УзелМен.ПДЗ;
			
			СтрМен.АКБМесяц = УзелМен.АКБМесяц;
			СтрМен.НовыеКлиенты = УзелМен.НовыеКлиенты;
			СтрМен.НовыеСетевыеПозиции = УзелМен.НовыеСетевыеПозиции;
			
			СтрМен.АКБМесяц_Подр = УзелПодр.АКБМесяц;
			СтрМен.НовыеКлиенты_Подр = УзелПодр.НовыеКлиенты;
			СтрМен.НовыеСетевыеПозиции_Подр = УзелПодр.НовыеСетевыеПозиции;
		КонецЦикла;
	КонецЦикла;
	
	ТабРезультат.Сортировать("ПодразделениеНаименование ВОЗР, МенеджерНаименование ВОЗР");
	Возврат ТабРезультат;
КонецФункции

Функция ПолучитьУзелПодразделения(Дерево, Подразделение)
	Если НЕ ЭтоЦелевоеПодразделение(Подразделение) Тогда
		Возврат Неопределено;
	КонецЕсли;
	
	Если Дерево.Получить(Подразделение) = Неопределено Тогда
		Узел = Новый Структура;
		Узел.Вставить("Подразделение", Подразделение);
		Узел.Вставить("ПродажиРубПлан", 0);
		Узел.Вставить("ПродажиРубФакт", 0);
		Узел.Вставить("ПродажиКгПлан", 0);
		Узел.Вставить("ПродажиКгФакт", 0);
		Узел.Вставить("Себестоимость", 0);
		Узел.Вставить("ВаловаяПрибыль", 0);
		Узел.Вставить("ЛогистикаФакт", 0);
		Узел.Вставить("НовыеКлиенты", 0);
		Узел.Вставить("НовыеСетевыеПозиции", 0);
		Узел.Вставить("АКБМесяц", 0);
		Узел.Вставить("ПДЗ", 0);
		Узел.Вставить("Менеджеры", Новый Соответствие);
		Дерево.Вставить(Подразделение, Узел);
	КонецЕсли;
	Возврат Дерево.Получить(Подразделение);
КонецФункции

Функция ПолучитьУзелМенеджера(УзелПодр, Менеджер)
	Если УзелПодр.Менеджеры.Получить(Менеджер) = Неопределено Тогда
		Узел = Новый Структура;
		Узел.Вставить("Менеджер", Менеджер);
		Узел.Вставить("ПродажиРубПлан", 0);
		Узел.Вставить("ПродажиРубФакт", 0);
		Узел.Вставить("ПродажиКгПлан", 0);
		Узел.Вставить("ПродажиКгФакт", 0);
		Узел.Вставить("Себестоимость", 0);
		Узел.Вставить("ВаловаяПрибыль", 0);
		Узел.Вставить("ЛогистикаФакт", 0);
		Узел.Вставить("НовыеКлиенты", 0);
		Узел.Вставить("НовыеСетевыеПозиции", 0);
		Узел.Вставить("АКБМесяц", 0);
		Узел.Вставить("ПДЗ", 0);
		УзелПодр.Менеджеры.Вставить(Менеджер, Узел);
	КонецЕсли;
	Возврат УзелПодр.Менеджеры.Получить(Менеджер);
КонецФункции

Процедура ЗаполнитьНовыеСетевыеПозиции(ДатаНач, ДатаКон, Дерево)
	ТабТек = ЗапросSKUСетевогоОтдела(ДатаНач, ДатаКон);
	ТабИст = ЗапросSKUСетевогоОтдела(НачалоДня(ДобавитьМесяц(ДатаНач, -24)), НачалоДня(ДатаНач) - 1);
	
	ИсторияSKU = Новый Соответствие;
	Для Каждого Стр Из ТабИст Цикл
		Если ЗначениеЗаполнено(Стр.Номенклатура) Тогда
			ИсторияSKU.Вставить(Стр.Номенклатура, Истина);
		КонецЕсли;
	КонецЦикла;
	
	УчетМенSKU = Новый Соответствие;
	УчетПодрSKU = Новый Соответствие;
	
	Для Каждого Стр Из ТабТек Цикл
		Если Не ЗначениеЗаполнено(Стр.Номенклатура) Или ИсторияSKU.Получить(Стр.Номенклатура) <> Неопределено Тогда
			Продолжить;
		КонецЕсли;
		
		УзелПодр = ПолучитьУзелПодразделения(Дерево, Стр.Подразделение);
		Если УзелПодр = Неопределено Тогда Продолжить; КонецЕсли;
		УзелМен = ПолучитьУзелМенеджера(УзелПодр, Стр.Менеджер);
		
		КлючМен = Строка(Стр.Менеджер) + "|" + Строка(Стр.Номенклатура);
		Если УчетМенSKU.Получить(КлючМен) = Неопределено Тогда
			УчетМенSKU.Вставить(КлючМен, Истина);
			УзелМен.НовыеСетевыеПозиции = УзелМен.НовыеСетевыеПозиции + 1;
		КонецЕсли;
		
		КлючПодр = Строка(Стр.Подразделение) + "|" + Строка(Стр.Номенклатура);
		Если УчетПодрSKU.Получить(КлючПодр) = Неопределено Тогда
			УчетПодрSKU.Вставить(КлючПодр, Истина);
			УзелПодр.НовыеСетевыеПозиции = УзелПодр.НовыеСетевыеПозиции + 1;
		КонецЕсли;
	КонецЦикла;

Функция ЭтоЦелевоеПодразделение(Подразделение)
	Если Подразделение = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;
	НаименованиеПодразделения = НРег(Строка(Подразделение));
	Если Найти(НаименованиеПодразделения, "консерв") > 0 Тогда
		Возврат Истина;
	КонецЕсли;
	Если Найти(НаименованиеПодразделения, "пф мяс") > 0 Тогда
		Возврат Истина;
	КонецЕсли;
	Если Найти(НаименованиеПодразделения, "пф рыб") > 0 Тогда
		Возврат Истина;
	КонецЕсли;
	Если Найти(НаименованиеПодразделения, "сетев") > 0 Тогда
		Возврат Истина;
	КонецЕсли;
	Возврат Ложь;

Функция РассчитатьПДЗПоМенеджерам(ДатаСреза, Кэш)
	ТабИтог = Новый ТаблицаЗначений;
	ТабИтог.Колонки.Добавить("Менеджер");
	ТабИтог.Колонки.Добавить("Подразделение");
	ТабИтог.Колонки.Добавить("ПДЗ", Новый ОписаниеТипов("Число"));
	
	ПДЗПоМенеджеру = Новый Соответствие;
	ПодрПоМенеджеру = Новый Соответствие;
	
	Для Каждого СтрЗаказ Из Кэш.Заказы Цикл
		Если СтрЗаказ.ДатаЗаказа < НачалоДня(ДобавитьМесяц(ДатаСреза, -6)) Или СтрЗаказ.ДатаЗаказа > КонецДня(ДатаСреза) Тогда
			Продолжить;
		КонецЕсли;
		
		СуммаЗаказа = СтрЗаказ.СуммаЗаказа;
		Если СуммаЗаказа = 0 Тогда
			Продолжить;
		КонецЕсли;
		
		ПП = Кэш.ПППоЗаказу.Получить(СтрЗаказ.Заказ);
		Если ТипЗнч(ПП) <> Тип("Дата") Тогда
			Продолжить;
		КонецЕсли;
		
		РазницаСекунд = НачалоДня(ДатаСреза) - НачалоДня(ПП);
		ДниПросрочки = Цел(РазницаСекунд / 86400);
		Если ДниПросрочки < 10 Тогда
			Продолжить;
		КонецЕсли;
		
		СуммаБезнал = 0;
		СуммаНал = 0;
		СуммаВзаим = 0;
		Оплаты = Кэш.ОплатыПоЗаказу.Получить(СтрЗаказ.Заказ);
		Если Оплаты <> Неопределено Тогда
			Для Каждого СтрОпл Из Оплаты Цикл
				Если СтрОпл.ДатаОплаты <= ДатаСреза Тогда
					Сум = НормЧисло(СтрОпл.Сумма);
					Если СтрОпл.ТипОплаты = 1 Тогда
						СуммаБезнал = СуммаБезнал + Сум;
					ИначеЕсли СтрОпл.ТипОплаты = 2 Тогда
						СуммаНал = СуммаНал + Сум;
					Иначе
						СуммаВзаим = СуммаВзаим + Сум;
					КонецЕсли;
				КонецЕсли;
			КонецЦикла;
		КонецЕсли;
		
		Оплачено = СуммаБезнал + СуммаНал + СуммаВзаим;
		Остаток = СуммаЗаказа - Оплачено;
		Если Остаток <= 0 Тогда
			Продолжить;
		КонецЕсли;
		
		ОстатокПроц = (Остаток / СуммаЗаказа) * 100;
		Если ОстатокПроц <= 0 Тогда
			Продолжить;
		КонецЕсли;
		
		ТекПДЗ = ПДЗПоМенеджеру.Получить(СтрЗаказ.Менеджер);
		Если ТекПДЗ = Неопределено Тогда
			ТекПДЗ = 0;
		КонецЕсли;
		ПДЗПоМенеджеру.Вставить(СтрЗаказ.Менеджер, ТекПДЗ + Остаток);
		Если ПодрПоМенеджеру.Получить(СтрЗаказ.Менеджер) = Неопределено Тогда
			ПодрПоМенеджеру.Вставить(СтрЗаказ.Менеджер, СтрЗаказ.Менеджер.Подразделение);
		КонецЕсли;
	КонецЦикла;
	
	Для Каждого Пара Из ПДЗПоМенеджеру Цикл
		Стр = ТабИтог.Добавить();
		Стр.Менеджер = Пара.Ключ;
		Стр.Подразделение = ПодрПоМенеджеру.Получить(Пара.Ключ);
		Стр.ПДЗ = Окр(Пара.Значение, 2);
	КонецЦикла;
	
	Возврат ТабИтог;
